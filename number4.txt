namespace Number4
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            int[] numbers = { 10, 20, 30, 40, 50 };
            label1.Text = "";
            for (int i = 0; i < numbers.Length; i++)
            {
                label1.Text += numbers[i].ToString() + " ";
            }

        }
    }
}
